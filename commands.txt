- To check cluster health
bin/kafka-metadata-quorum.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 describe --status

- Create a topic
bin/kafka-topics.sh --create --topic my_production_topic \
  --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --partitions 3 \
  --replication-factor 3 \
  --config min.insync.replicas=2 \
  --config retention.bytes=5368709120 \
  --config retention.ms=86400000

- List the topics
bin/kafka-topics.sh --list --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092

- Delete the topic
bin/kafka-topics.sh --delete --topic topic_name \
  --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092

- To check data in topic
bin/kafka-console-consumer.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --topic topic_name \
  --from-beginning

- To describe a topic
bin/kafka-log-dirs.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --describe --topic-list topic_name
- To decrible a consumer
bin/kafka-consumer-groups.sh \
  --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --describe \
  --group python_consumer

- To see all consumers
bin/kafka-consumer-groups.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 --list

- To delete all consumers
bin/kafka-consumer-groups.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --delete --all-groups

- To reset offsets to earliest
bin/kafka-consumer-groups.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --group your-group-name \
  --reset-offsets \
  --to-earliest \
  --topic your-topic \
  --execute

- To reset offsets to latest
bin/kafka-consumer-groups.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --group your-group-name \
  --reset-offsets \
  --to-latest \
  --topic your-topic \
  --execute

- To reset offsets for all topics
bin/kafka-consumer-groups.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --group your-group-name \
  --reset-offsets \
  --to-earliest \
  --all \
  --execute

- To delete a particular consumer group
bin/kafka-consumer-groups.sh --bootstrap-server kafka01:9092,kafka02:9092,kafka03:9092 \
  --delete --group my-consumer
